@echo off
title Blocage Windows Update
color 0C

echo ==========================================
echo   BLOCAGE DE WINDOWS UPDATE
echo ==========================================
echo.

:: Arret des services
echo [1/4] Arret des services...

net stop wuauserv /y
net stop bits /y
net stop usosvc /y
net stop WaaSMedicSvc /y

:: Desactivation des services
echo.
echo [2/4] Desactivation des services...

sc config wuauserv start= disabled
sc config bits start= disabled
sc config usosvc start= disabled

:: Blocage du service Windows Update dans le registre
echo.
echo [3/4] Blocage de Windows Update...

reg add "HKLM\SYSTEM\CurrentControlSet\Services\wuauserv" /v Start /t REG_DWORD /d 4 /f

:: Desactivation de plusieurs taches Windows Update
echo.
echo [4/4] Desactivation des taches planifiees...

schtasks /Change /TN "\Microsoft\Windows\WindowsUpdate\Scheduled Start" /Disable
schtasks /Change /TN "\Microsoft\Windows\UpdateOrchestrator\Schedule Scan" /Disable
schtasks /Change /TN "\Microsoft\Windows\UpdateOrchestrator\Schedule Scan Static Task" /Disable
schtasks /Change /TN "\Microsoft\Windows\UpdateOrchestrator\USO_UxBroker" /Disable
schtasks /Change /TN "\Microsoft\Windows\UpdateOrchestrator\UpdateModelTask" /Disable
schtasks /Change /TN "\Microsoft\Windows\WaaSMedic\PerformRemediation" /Disable

echo.
echo ==========================================
echo   WINDOWS UPDATE A ETE BLOQUE
echo ==========================================
echo.
pause
